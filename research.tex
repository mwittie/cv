\subsection*{Software Systems Released}
\hrule
\vspace{0.2cm}

\begin{itemize}
    \item Blocky Attestation Service~(AS)
    \hfill {\it June 2025} \\
    WebAssembly~(WASM) based verifiable serverless compute platform on Trusted Execution Environments~(TEEs).
    Available at \url{https://docs.blocky.rocks}.

    \item Roadkill Observation and Data System~(ROaDS)
    \hfill {\it March 2021}\\
    A mobile app reporting system for roadkill observations developed for the Federal Highway Administration~(FHWA).

    \item Unstable Slope Measurement Program (USMP)
    \hfill {\it May 2017}\\
    An inventory system for unstable earth slopes next to roadways and trails.\\
    Available at \url{http://usmp.info}.

    \item MITATE: Mobile Internet Testbed for Application Traffic Experimentation
    \hfill {\it pre-launch}\\
    A Google M-Lab tool for traffic experimentation in cellular networks. \\
    Available at \url{http://github.com/msu-netlab/mitate}.

    \item \texttt{pcp}: Ping through CDN Proxies
    \hfill {\it August 2015}\\
    A client tool for latency estimation based on client to CDN replica server and inter-replica network measurement. \\
    Available at \url{http://github.com/msu-netlab/pcp}.

    \item \texttt{dp}: DNS Proxy
    \hfill {\it August 2015}\\
    A tool for client side selection of resolutions from multiple DNS servers.\\
    Available at \url{http://github.com/msu-netlab/dp}.

    \item NWP OTIIS: Heterogeneous Data Integration for Operations and Travel Information Sharing
    \hfill {\it August 2014}\\
    A road conditions information site for travelers in the I-90/94 corridor.\\
    Available at \url{http://roadstosafediscovery.com}.

\end{itemize}
